# APIC NX-OS style CLI concept example
# Validate exact syntax on the installed APIC 6.2(x) release.

configure

vlan-domain dom-LAB
  vlan 100-199

tenant LAB
  vrf context VRF-LAB

  bridge-domain BD-WEB
    vrf member VRF-LAB

  bridge-domain BD-APP
    vrf member VRF-LAB

  interface bridge-domain BD-WEB
    ip address 10.10.10.1/24
  exit

  interface bridge-domain BD-APP
    ip address 10.10.20.1/24
  exit

  application APP-LAB
    epg EPG-WEB
      bridge-domain member BD-WEB
    exit
    epg EPG-APP
      bridge-domain member BD-APP
    exit
  exit
exit

# Principle:
# access VLAN -> EPG classification -> BD -> VRF
